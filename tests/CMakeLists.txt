# Assertions (and so death tests) exist in builds without NDEBUG: Debug with single-config
# generators, which is all the presets use.
if(CMAKE_BUILD_TYPE STREQUAL "Debug")
    set(AX_TESTS_HAVE_ASSERTS ON)
else()
    set(AX_TESTS_HAVE_ASSERTS OFF)
endif()

# A death test passes by aborting, so a sanitizer report must not abort inside one or it would pass
# for the wrong reason (macOS sanitizers abort by default). These settings make a report end the
# process with the sanitizer's non-zero exit code instead, failing the test (decision 41). They
# replace the test presets' sanitizer options for death tests only.
set(AX_DEATH_TEST_ENVIRONMENT
    "ASAN_OPTIONS=abort_on_error=0:halt_on_error=1:strict_string_checks=1"
    "UBSAN_OPTIONS=abort_on_error=0:halt_on_error=1:print_stacktrace=1"
    "TSAN_OPTIONS=abort_on_error=0:halt_on_error=1")

# ax_add_unit_test(<module> <source>)
#   <source> is relative to the calling CMakeLists.txt.
#   Builds one executable per test file and registers each AX_TEST(<case>) in it as the ctest
#   test <module>.<file>.<case>, where <file> is the source name without its "test_" prefix.
#   The executable's main() is generated from the same scan, so cases are never listed by hand.
function(ax_add_unit_test module source)
    get_filename_component(_file "${source}" NAME_WE)
    string(REGEX REPLACE "^test_" "" _file "${_file}")
    set(_suite "${module}.${_file}")
    set(_exe "test_${module}_${_file}")
    set(_src "${CMAKE_CURRENT_SOURCE_DIR}/${source}")

    # Re-run the scan whenever the test file changes.
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${_src}")
    file(STRINGS "${_src}" _lines REGEX "^AX_(DEATH_)?TEST\\([A-Za-z0-9_]+\\)")
    set(_cases "")
    set(_death_cases "")
    foreach(_line IN LISTS _lines)
        string(REGEX REPLACE "^AX_(DEATH_)?TEST\\(([A-Za-z0-9_]+)\\).*" "\\2" _case "${_line}")
        list(APPEND _cases "${_case}")
        if(_line MATCHES "^AX_DEATH_TEST")
            list(APPEND _death_cases "${_case}")
        endif()
    endforeach()
    if(NOT _cases)
        message(FATAL_ERROR "${source}: no AX_TEST(...) cases found")
    endif()
    list(SORT _cases)
    list(REMOVE_DUPLICATES _cases)

    set(_decls "")
    set(_table "")
    foreach(_case IN LISTS _cases)
        if(_case IN_LIST _death_cases)
            set(_is_death 1)
        else()
            set(_is_death 0)
        endif()
        string(APPEND _decls "void ax_test__${_case}(void);\n")
        string(APPEND _table "    {\"${_case}\", ax_test__${_case}, ${_is_death}},\n")
    endforeach()
    set(_main "${CMAKE_CURRENT_BINARY_DIR}/generated/${_exe}_main.c")
    file(CONFIGURE OUTPUT "${_main}" @ONLY CONTENT
"/* Generated by tests/CMakeLists.txt from ${source}. Do not edit. */
#include \"ax_test.h\"

@_decls@
static const ax_test_case ax_test_cases[] = {
@_table@};

int main(int argc, char **argv)
{
    return ax_test_main(\"@_suite@\", ax_test_cases,
                        sizeof ax_test_cases / sizeof ax_test_cases[0], argc, argv);
}
")

    add_executable(${_exe} "${_src}" "${_main}")
    target_link_libraries(${_exe} PRIVATE ax_test_harness ax_base)
    ax_target_defaults(${_exe})
    if(EMSCRIPTEN)
        # Tests in Node: a heap that grows past Emscripten's default 16 MB.
        target_link_options(${_exe} PRIVATE "-sALLOW_MEMORY_GROWTH=1")
    endif()

    foreach(_case IN LISTS _cases)
        if(_case IN_LIST _death_cases)
            # Death tests need the assertions they trigger: only in builds that keep them.
            if(NOT AX_TESTS_HAVE_ASSERTS)
                continue()
            endif()
            add_test(NAME "${_suite}.${_case}" COMMAND ${_exe} ${_case})
            set_tests_properties("${_suite}.${_case}" PROPERTIES
                LABELS "${module};death" TIMEOUT 60 SKIP_RETURN_CODE 77
                ENVIRONMENT "${AX_DEATH_TEST_ENVIRONMENT}")
        else()
            add_test(NAME "${_suite}.${_case}" COMMAND ${_exe} ${_case})
            set_tests_properties("${_suite}.${_case}" PROPERTIES
                LABELS "${module}" TIMEOUT 60 SKIP_RETURN_CODE 77)
        endif()
    endforeach()
endfunction()

# Every header must compile on its own (includes what it uses). One generated translation unit
# per header, built with the same strict settings; failure breaks the build.
set(_ax_header_modules base core)
if(AX_BUILD_PLATFORM)
    list(APPEND _ax_header_modules platform)
endif()
if(AX_BUILD_GPU)
    list(APPEND _ax_header_modules gpu)
endif()
set(_ax_headers "")
foreach(_m IN LISTS _ax_header_modules)
    file(GLOB _m_headers RELATIVE "${PROJECT_SOURCE_DIR}/${_m}/include"
         CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/${_m}/include/axion/*.h"
                           "${PROJECT_SOURCE_DIR}/${_m}/include/axion/${_m}/*.h")
    list(APPEND _ax_headers ${_m_headers})
endforeach()
file(GLOB _ir_headers RELATIVE "${PROJECT_SOURCE_DIR}/render/ir/include"
     CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/render/ir/include/axion/render/*.h")
list(APPEND _ax_headers ${_ir_headers})
file(GLOB _rs_headers RELATIVE "${PROJECT_SOURCE_DIR}/render/soft/include"
     CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/render/soft/include/axion/render/*.h")
list(APPEND _ax_headers ${_rs_headers})
if(AX_BUILD_PLATFORM)
    file(GLOB _rf_headers RELATIVE "${PROJECT_SOURCE_DIR}/render/frame/include"
         CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/render/frame/include/axion/render/*.h")
    list(APPEND _ax_headers ${_rf_headers})
endif()
if(AX_BUILD_GPU)
    file(GLOB _rw_headers RELATIVE "${PROJECT_SOURCE_DIR}/render/wgpu/include"
         CONFIGURE_DEPENDS "${PROJECT_SOURCE_DIR}/render/wgpu/include/axion/render/*.h")
    list(APPEND _ax_headers ${_rw_headers})
endif()
if(NOT AX_BUILD_PLATFORM)
    # These need the platform module's SDK.
    list(REMOVE_ITEM _ax_headers "axion/gpu/ax_gpu_surface.h" "axion/render/ax_renderer.h"
         "axion/render/ax_render_wgpu_surface.h")
endif()
set(_ax_header_tus "")
foreach(_h IN LISTS _ax_headers)
    string(MAKE_C_IDENTIFIER "${_h}" _id)
    set(_tu "${CMAKE_CURRENT_BINARY_DIR}/generated/header_check/${_id}.c")
    file(CONFIGURE OUTPUT "${_tu}" CONTENT "#include \"${_h}\"\n#include \"${_h}\"\ntypedef int ax_header_check_${_id};\n")
    list(APPEND _ax_header_tus "${_tu}")
endforeach()
add_library(ax_header_check OBJECT ${_ax_header_tus})
target_link_libraries(ax_header_check PRIVATE ax_base ax_core ax_render_ir ax_render_soft)
if(AX_BUILD_PLATFORM)
    target_link_libraries(ax_header_check PRIVATE ax_platform ax_render_frame)
endif()
if(AX_BUILD_GPU)
    target_link_libraries(ax_header_check PRIVATE ax_gpu ax_render_wgpu)
endif()
ax_target_defaults(ax_header_check)
